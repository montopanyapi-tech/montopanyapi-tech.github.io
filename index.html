// TASARIMI BİREBİR EKRANDAKİ ÖLÇEKLE İNDİRME FONKSİYONU function downloadDesign() { const baseImg = document.getElementById("base-house"); if (!baseImg.complete || baseImg.naturalWidth === 0) { alert("Demo ev resmi yüklenemedi!"); return; } const canvas = document.createElement("canvas"); const ctx = canvas.getContext("2d"); // Canvas boyutlarını orijinal resim kalitesine eşitle canvas.width = baseImg.naturalWidth; canvas.height = baseImg.naturalHeight; // 1. Arka plandaki evi çiz ctx.drawImage(baseImg, 0, 0, canvas.width, canvas.height); // 2. Seçilen desen varsa ekrandaki ölçek oranına göre çiz if (activePanelDataUrl) { const patternImg = new Image(); patternImg.onload = function() { ctx.globalCompositeOperation = "multiply"; ctx.globalAlpha = 0.95; // Ekrandaki 180px ölçeğini orijinal resim çözünürlüğüne oranlıyoruz const displayWidth = baseImg.clientWidth || 1200; const scaleFactor = canvas.width / displayWidth; const targetPatternWidth = 180 * scaleFactor; // Ekrandaki 180px karşılığı // Deseni ekrandaki boyuta ölçeklendirmek için geçici Canvas const patternCanvas = document.createElement("canvas"); const patternCtx = patternCanvas.getContext("2d"); const aspect = patternImg.height / patternImg.width; patternCanvas.width = targetPatternWidth; patternCanvas.height = targetPatternWidth * aspect; patternCtx.drawImage(patternImg, 0, 0, patternCanvas.width, patternCanvas.height); const pattern = ctx.createPattern(patternCanvas, "repeat"); ctx.fillStyle = pattern; const scaleX = canvas.width / 100; const scaleY = canvas.height / 100; // SOL DUVAR (Pencereler Kırpılmış Detaylı Poligon) ctx.beginPath(); // Dış sınır ctx.moveTo(10.0 * scaleX, 34.6 * scaleY); ctx.lineTo(50.8 * scaleX, 34.6 * scaleY); ctx.lineTo(50.8 * scaleX, 65.4 * scaleY); ctx.lineTo(10.0 * scaleX, 65.4 * scaleY); ctx.lineTo(10.0 * scaleX, 34.6 * scaleY); // Sol pencere deliği ctx.lineTo(18.1 * scaleX, 39.5 * scaleY); ctx.lineTo(18.1 * scaleX, 56.4 * scaleY); ctx.lineTo(26.6 * scaleX, 56.4 * scaleY); ctx.lineTo(26.6 * scaleX, 39.5 * scaleY); ctx.lineTo(18.1 * scaleX, 39.5 * scaleY); ctx.lineTo(10.0 * scaleX, 34.6 * scaleY); // Sağ pencere deliği ctx.lineTo(34.7 * scaleX, 39.5 * scaleY); ctx.lineTo(34.7 * scaleX, 56.4 * scaleY); ctx.lineTo(43.1 * scaleX, 56.4 * scaleY); ctx.lineTo(43.1 * scaleX, 39.5 * scaleY); ctx.lineTo(34.7 * scaleX, 39.5 * scaleY); ctx.closePath(); ctx.fill(); // SAĞ VERANDA DUVARI (Kapı Kırpılmış) ctx.beginPath(); ctx.moveTo(64.2 * scaleX, 38.0 * scaleY); ctx.lineTo(91.6 * scaleX, 38.0 * scaleY); ctx.lineTo(91.6 * scaleX, 63.6 * scaleY); ctx.lineTo(64.2 * scaleX, 63.6 * scaleY); ctx.lineTo(64.2 * scaleX, 38.0 * scaleY); // Veranda kapı deliği ctx.lineTo(68.4 * scaleX, 41.5 * scaleY); ctx.lineTo(68.4 * scaleX, 62.4 * scaleY); ctx.lineTo(80.3 * scaleX, 62.4 * scaleY); ctx.lineTo(80.3 * scaleX, 41.5 * scaleY); ctx.closePath(); ctx.fill(); triggerDownload(canvas); }; patternImg.src = activePanelDataUrl; } else { triggerDownload(canvas); } }
